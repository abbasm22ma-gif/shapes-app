<div class="app">
  <div class="toolbar">
    <button class="copy-btn" id="copySelectedBtn" title="Duplicate selected shape (Ctrl+D / ⌘D)">
      📋 Copy selected
    </button>
    <button class="reset-btn" id="resetBtn" title="Reset to initial shapes">
      ⟲ Reset scene
    </button>
    <span class="hint">click to select · drag to move · Ctrl/Cmd + D to copy</span>

  <canvas id="shapeCanvas" width="800" height="520"></canvas>

<script>
  (function(){
    const canvas = document.getElementById('shapeCanvas');
    const ctx = canvas.getContext('2d');

    // ---------- shape data ----------
    let shapes = [];
    let selectedId = null;

    // unique id generator
    let nextId = 1;
    function genId() {
      return nextId++;
    }

    // ---------- helpers ----------
    // distance from point to shape (only needed for selection)
    // circle: center (x,y), radius r
    function isPointInCircle(px, py, cx, cy, r) {
      const dx = px - cx;
      const dy = py - cy;
      return dx * dx + dy * dy <= r * r;
    }

    // rectangle: x, y, w, h  (top-left)
    function isPointInRect(px, py, rx, ry, rw, rh) {
      return px >= rx && px <= rx + rw && py >= ry && py <= ry + rh;
    }

    // triangle: vertices (x1,y1), (x2,y2), (x3,y3) — using barycentric or area method
    function isPointInTriangle(px, py, ax, ay, bx, by, cx, cy) {
      const areaABC = Math.abs((bx - ax) * (cy - ay) - (cx - ax) * (by - ay));
      if (areaABC < 0.0001) return false; // degenerate
      const areaPAB = Math.abs((ax - px) * (by - py) - (bx - px) * (ay - py));
      const areaPBC = Math.abs((bx - px) * (cy - py) - (cx - px) * (by - py));
      const areaPCA = Math.abs((cx - px) * (ay - py) - (ax - px) * (cy - py));
      const sum = areaPAB + areaPBC + areaPCA;
      return Math.abs(sum - areaABC) < 0.001;
    }

    // find shape under cursor (topmost first, because we iterate from end)
    function getShapeAt(x, y) {
      for (let i = shapes.length - 1; i >= 0; i--) {
        const s = shapes[i];
        if (s.type === 'circle') {
          if (isPointInCircle(x, y, s.cx, s.cy, s.r)) return s;
        } else if (s.type === 'rect') {
          if (isPointInRect(x, y, s.x, s.y, s.w, s.h)) return s;
        } else if (s.type === 'triangle') {
          if (isPointInTriangle(x, y, s.x1, s.y1, s.x2, s.y2, s.x3, s.y3)) return s;
        }
      }
      return null;
    }

    // ---------- move shape by dx, dy ----------
    function moveShape(shape, dx, dy) {
      if (shape.type === 'circle') {
        shape.cx += dx;
        shape.cy += dy;
      } else if (shape.type === 'rect') {
        shape.x += dx;
        shape.y += dy;
      } else if (shape.type === 'triangle') {
        shape.x1 += dx; shape.y1 += dy;
        shape.x2 += dx; shape.y2 += dy;
        shape.x3 += dx; shape.y3 += dy;
      }
    }

    // ---------- copy shape (deep clone) ----------
    function copyShape(shape) {
      const clone = JSON.parse(JSON.stringify(shape)); // simple safe copy
      clone.id = genId();
      // offset a bit so it's visible and not exactly on top
      const OFFSET = 24;
      if (clone.type === 'circle') {
        clone.cx += OFFSET;
        clone.cy += OFFSET;
      } else if (clone.type === 'rect') {
        clone.x += OFFSET;
        clone.y += OFFSET;
      } else if (clone.type === 'triangle') {
        clone.x1 += OFFSET; clone.y1 += OFFSET;
        clone.x2 += OFFSET; clone.y2 += OFFSET;
        clone.x3 += OFFSET; clone.y3 += OFFSET;
      }
      return clone;
    }

    // ---------- add shape to scene + auto-select ----------
    function addShape(shape) {
      shapes.push(shape);
      selectedId = shape.id;
    }

    // ---------- initial shapes (colourful mix) ----------
    function resetScene() {
      shapes = [];
      selectedId = null;

      // circle (blue)
      addShape({
        id: genId(),
        type: 'circle',
        cx: 150, cy: 140, r: 60,
        fill: '#3b82f6', stroke: '#1e3a8a', lineWidth: 3
      });

      // rectangle (amber)
      addShape({
        id: genId(),
        type: 'rect',
        x: 440, y: 60, w: 140, h: 100,
        fill: '#f59e0b', stroke: '#b45309', lineWidth: 3
      });

      // triangle (emerald)
      addShape({
        id: genId(),
        type: 'triangle',
        x1: 660, y1: 310,
        x2: 580, y2: 430,
        x3: 740, y3: 430,
        fill: '#10b981', stroke: '#065f46', lineWidth: 3
      });

      // extra circle (pink) to make it richer
      addShape({
        id: genId(),
        type: 'circle',
        cx: 310, cy: 380, r: 50,
        fill: '#ec4899', stroke: '#9d174d', lineWidth: 3
      });

      // extra rectangle (purple)
      addShape({
        id: genId(),
        type: 'rect',
        x: 100, y: 320, w: 110, h: 80,
        fill: '#8b5cf6', stroke: '#5b21b6', lineWidth: 3
      });

      // one more triangle (orange) for fun
      addShape({
        id: genId(),
        type: 'triangle',
        x1: 230, y1: 60,
        x2: 150, y2: 190,
        x3: 310, y3: 190,
        fill: '#f97316', stroke: '#9a3412', lineWidth: 3
      });

      // no shape selected after reset (optional: keep last selected)
      selectedId = null;
      draw();
    }

    // ---------- drawing ----------
    function drawShape(shape, isSelected) {
      ctx.save();

      // draw fill & stroke
      if (shape.type === 'circle') {
        ctx.beginPath();
        ctx.arc(shape.cx, shape.cy, shape.r, 0, Math.PI * 2);
        ctx.fillStyle = shape.fill;
        ctx.fill();
        ctx.lineWidth = shape.lineWidth;
        ctx.strokeStyle = shape.stroke;
        ctx.stroke();
      } else if (shape.type === 'rect') {
        ctx.beginPath();
        ctx.rect(shape.x, shape.y, shape.w, shape.h);
        ctx.fillStyle = shape.fill;
        ctx.fill();
        ctx.lineWidth = shape.lineWidth;
        ctx.strokeStyle = shape.stroke;
        ctx.stroke();
      } else if (shape.type === 'triangle') {
        ctx.beginPath();
        ctx.moveTo(shape.x1, shape.y1);
        ctx.lineTo(shape.x2, shape.y2);
        ctx.lineTo(shape.x3, shape.y3);
        ctx.closePath();
        ctx.fillStyle = shape.fill;
        ctx.fill();
        ctx.lineWidth = shape.lineWidth;
        ctx.strokeStyle = shape.stroke;
        ctx.stroke();
      }

      // selection indicator (thick dashed ring / glow)
      if (isSelected) {
        ctx.shadowColor = '#ffffff';
        ctx.shadowBlur = 18;
        ctx.shadowOffsetX = 0;
        ctx.shadowOffsetY = 0;
        // draw a second outline on top
        ctx.lineWidth = 5;
        ctx.strokeStyle = '#ffffff';
        ctx.setLineDash([8, 8]);
        if (shape.type === 'circle') {
          ctx.beginPath();
          ctx.arc(shape.cx, shape.cy, shape.r + 3, 0, Math.PI * 2);
          ctx.stroke();
        } else if (shape.type === 'rect') {
          ctx.strokeRect(shape.x - 2, shape.y - 2, shape.w + 4, shape.h + 4);
        } else if (shape.type === 'triangle') {
          ctx.beginPath();
          ctx.moveTo(shape.x1, shape.y1);
          ctx.lineTo(shape.x2, shape.y2);
          ctx.lineTo(shape.x3, shape.y3);
          ctx.closePath();
          ctx.stroke();
        }
        ctx.setLineDash([]); // reset
      }

      ctx.restore();
    }

    function draw() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      // soft grid dots (background)
      ctx.save();
      ctx.fillStyle = '#cbd5e1';
      for (let x = 20; x < canvas.width; x += 40) {
        for (let y = 20; y < canvas.height; y += 40) {
          ctx.beginPath();
          ctx.arc(x, y, 1.8, 0, Math.PI * 2);
          ctx.fill();
        }
      }
      ctx.restore();

      // draw all shapes (back to front)
      for (const shape of shapes) {
        const isSelected = (shape.id === selectedId);
        drawShape(shape, isSelected);
      }
    }

    // ---------- mouse / touch drag logic ----------
    let isDragging = false;
    let dragStartX = 0, dragStartY = 0;
    let dragShapeId = null;

    // canvas coordinates from event
    function getCanvasCoords(e) {
      const rect = canvas.getBoundingClientRect();
      const scaleX = canvas.width / rect.width;   // canvas physical pixels / CSS pixels
      const scaleY = canvas.height / rect.height;

      let clientX, clientY;
      if (e.touches) {
        clientX = e.touches[0].clientX;
        clientY = e.touches[0].clientY;
        e.preventDefault();
      } else {
        clientX = e.clientX;
        clientY = e.clientY;
      }

      const canvasX = (clientX - rect.left) * scaleX;
      const canvasY = (clientY - rect.top) * scaleY;
      return { x: Math.min(Math.max(canvasX, 0), canvas.width), y: Math.min(Math.max(canvasY, 0), canvas.height) };
    }

    function onPointerDown(e) {
      e.preventDefault(); // prevent text selection / scrolling

      const { x, y } = getCanvasCoords(e);
      const shape = getShapeAt(x, y);

      if (shape) {
        // select it
        selectedId = shape.id;
        // start drag
        isDragging = true;
        dragShapeId = shape.id;
        dragStartX = x;
        dragStartY = y;
        draw();
      } else {
        // clicked empty area → deselect
        selectedId = null;
        isDragging = false;
        dragShapeId = null;
        draw();
      }
    }

    function onPointerMove(e) {
      if (!isDragging || dragShapeId === null) return;
      e.preventDefault();

      const { x, y } = getCanvasCoords(e);

      // find the shape being dragged (by id)
      const shape = shapes.find(s => s.id === dragShapeId);
      if (!shape) return;

      // compute delta
      const dx = x - dragStartX;
      const dy = y - dragStartY;

      if (dx === 0 && dy === 0) return;

      // move it
      moveShape(shape, dx, dy);

      // update drag start for next move
      dragStartX = x;
      dragStartY = y;

      draw();
    }

    function onPointerUp(e) {
      if (isDragging) {
        // stop dragging, keep selection
        isDragging = false;
        dragShapeId = null;
        // no redraw needed – already drawn, but just in case
        draw();
      }
    }

    // ---------- copy selected shape ----------
    function copySelectedShape() {
      if (selectedId === null) {
        // optional: give feedback with a subtle flash or just ignore
        return;
      }
      const original = shapes.find(s => s.id === selectedId);
      if (!original) return;

      const clone = copyShape(original);
      shapes.push(clone);
      selectedId = clone.id;   // select the new copy
      draw();
    }

    // ---------- event listeners ----------
    canvas.addEventListener('mousedown', onPointerDown);
    canvas.addEventListener('mousemove', onPointerMove);
    canvas.addEventListener('mouseup', onPointerUp);
    canvas.addEventListener('mouseleave', (e) => {
      // if mouse leaves canvas while dragging, treat as mouseup to prevent stuck drag
      if (isDragging) {
        isDragging = false;
        dragShapeId = null;
        draw();
      }
    });

    // touch events (passive: false to allow preventDefault)
    canvas.addEventListener('touchstart', onPointerDown, { passive: false });
    canvas.addEventListener('touchmove', onPointerMove, { passive: false });
    canvas.addEventListener('touchend', onPointerUp);
    canvas.addEventListener('touchcancel', (e) => {
      if (isDragging) {
        isDragging = false;
        dragShapeId = null;
        draw();
      }
    });

    // keyboard shortcut: Ctrl+D / Cmd+D to copy selected
    window.addEventListener('keydown', (e) => {
      // Ctrl+D or Command+D
      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'd') {
        e.preventDefault();   // prevent browser bookmark
        copySelectedShape();
      }
    });

    // buttons
    document.getElementById('copySelectedBtn').addEventListener('click', copySelectedShape);
    document.getElementById('resetBtn').addEventListener('click', resetScene);

    // ---------- initial draw ----------
    resetScene();  // builds shapes and calls draw

    // also make sure selection is cleared after reset (already done in resetScene)
    // but we want a shape selected initially? let's select the first one for convenience
    if (shapes.length > 0) {
      selectedId = shapes[0].id; // select the blue circle
      draw();
    }
  })();
</script>
